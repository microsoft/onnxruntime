# Minimal deps for the onnxruntime_MATERIALIZE_ONNX_NODE_TESTS configure-time gate, which
# regenerates the ONNX node-test corpus into the build tree for onnx_test_runner.
# Install via `pip install -r` before configuring materialization on a bare agent.
#
# Deliberately kept to onnx + numpy ONLY so Dependabot can track these two pins (this file lives
# at the repo root, the sole pip directory Dependabot scans) WITHOUT bare-agent builds having
# to pull the heavy docker requirements.txt.
#
# PIN-LOCK: these values MUST stay in lockstep with the CMake gate (cmake/onnxruntime_unittests.cmake,
# which HARD-FATALs when the installed versions differ from its expected onnx/numpy pins) and with
# the Python MIN pins. Bump here and in cmake/deps.txt + the numpy MIN pins together.
onnx==1.23.2
numpy==2.2.6; python_version < "3.11"
numpy==2.4.2; python_version >= "3.11"
